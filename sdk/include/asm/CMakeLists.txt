set(OPT_MS "")

if(ARCH STREQUAL "i386")
    set(_filename ks386.inc)
elseif(ARCH STREQUAL "amd64")
    set(_filename ksamd64.inc)
elseif(ARCH STREQUAL "arm64")
    set(_filename ksarm64.h)
endif()

add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${_filename}
    COMMAND native-geninc ${CMAKE_CURRENT_BINARY_DIR}/genincdata.dll ${CMAKE_CURRENT_BINARY_DIR}/${_filename} ${OPT_MS}
    DEPENDS genincdata native-geninc)

add_custom_target(asm
    DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/${_filename})

# genincdata.dll is C, and cargo compiles and links it:
# sdk/lib/rust/genincdata-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# Generated inputs those compiles read, ordered before them.
add_library(genincdata_shims OBJECT genincdata.c)
add_dependencies(genincdata_shims bugcodes psdk xdk)
add_cargo_module(genincdata genincdata-cargo genincdata.dll
    KIND win32dll
    NO_CD
    EXTRA_DEPENDS genincdata_shims)
