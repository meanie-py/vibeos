# VibeOS Raspberry Pi 5 bench card: the normal boot. Deployed by tools/pi5/stage.sh
# on every `pi5 boot`, so edit it here, never on the card.
#
# A normal boot starts Raspberry Pi OS: the remote stager. It owns the board's
# Ethernet, which UEFI has no driver for, so `vibeos-cli pi5 boot` SSHes in, copies
# this tree's loader and ramdisk onto this partition, and runs `reboot '0 tryboot'`.
# For that ONE boot the firmware reads tryboot.txt INSTEAD of this file (a
# [tryboot] section here is never consulted: that filter is for autoboot.txt).
# Any later reset is Linux again.
arm_64bit=1
auto_initramfs=1
disable_overscan=1
enable_uart=1
