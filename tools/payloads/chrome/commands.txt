# Chrome's child processes do not write to the sweep's pipe, so the whole tree logs to a file
# that catfile prints afterwards (it runs even when Chrome is killed at the timeout).
timeout chrome\chrome.exe 420
# Look at the whole tree while the utility processes still wait for their connection.
sample chrome\chrome.exe 75
# --ipc-connection-timeout: a child that is not connected in 15 s exits, and one CPU under TCG is slow.
# A crash Crashpad catches never reaches the kernel's print, so dumpinfo reads its minidumps.
# Run M: multi-process, as Chrome runs by default. Dumps the page in about 2.5 minutes under TCG.
chrome\chrome.exe --headless --no-sandbox --disable-gpu --no-first-run --disable-breakpad --enable-logging --ipc-connection-timeout=120 --log-file=C:\TEMP\chrome_m.log --v=1 --user-data-dir=C:\TEMP\chrome-m --dump-dom file:///X:/reactos/bin/chrome-pages/hello.html
catfile\catfile.exe C:\TEMP\chrome_m.log
dumpinfo\dumpinfo.exe C:\TEMP\chrome-m\Crashpad\reports
# Run N (rung 2): the page over HTTP from servepage, in the guest; its script is a second fetch,
# so "net js ran: 42" means the network service fetched both and V8 ran the script.
# Windows' answer: servepage.win11.txt.
timeout servepage\servepage.exe 420
servepage\servepage.exe 8000 X:\reactos\bin\chrome\chrome.exe --headless --no-sandbox --disable-gpu --no-first-run --disable-breakpad --enable-logging --ipc-connection-timeout=120 --log-file=C:\TEMP\chrome_n.log --user-data-dir=C:\TEMP\chrome-n --dump-dom http://127.0.0.1:8000/
catfile\catfile.exe C:\TEMP\chrome_n.log
# Run I: a page off the internet through QEMU's NAT (needs DNS and the host's connectivity, so
# exploratory, not a pass/fail rung). Plain http://example.com/ failed ERR_BLOCKED_BY_CLIENT here
# while Windows' Chrome loads it, upgrading it to https first; so I1 asks for https outright and
# I2 asks for http with the upgrade off, to tell TLS apart from the plain fetch.
chrome\chrome.exe --headless --no-sandbox --disable-gpu --no-first-run --disable-breakpad --enable-logging --ipc-connection-timeout=120 --log-file=C:\TEMP\chrome_i1.log --user-data-dir=C:\TEMP\chrome-i1 --dump-dom https://example.com/
chrome\chrome.exe --headless --no-sandbox --disable-gpu --no-first-run --disable-breakpad --enable-logging --ipc-connection-timeout=120 --log-file=C:\TEMP\chrome_i2.log --disable-features=HttpsUpgrades --log-net-log=C:\TEMP\net_i2.json --user-data-dir=C:\TEMP\chrome-i2 --dump-dom http://example.com/
# The net log's events that failed ERR_BLOCKED_BY_CLIENT (-20), and every event naming the page.
# On Windows the same binary and flags load the page and log no -20 at all.
grepfile\grepfile.exe C:\TEMP\net_i2.json 120 :-20, :-20} example.com
# Run S: Run M with the sandbox on. Where it first breaks is the start of the sandbox quest.

catfile\catfile.exe C:\TEMP\chrome_s.log
