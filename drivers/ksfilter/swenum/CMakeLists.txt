# ============================================================================
# swenum.sys — cargo-driven final link (sdk/lib/rust/swenum-cargo)
# ============================================================================
# swenum.c (the KS software bus enumerator) is translated to Rust
# (sdk/lib/rust/swenum-rust, kept on disk per the never-delete rule). Replaces
# the MODULE link + rust_swenum custom_command. swenum-cargo's build.rs
# (reactos-link, ModuleKind::KernelDll, user-target body + i386 -sse,-sse2)
# imports ks (ks.sys — the Ks* bus-enum entry points) + ntoskrnl + hal and
# reuses memcmp.

# swenum.sys is all Rust (swenum-rust) and cargo links it:
# sdk/lib/rust/swenum-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The resources are the one thing cmake still compiles, in this OBJECT library
# whose windres command the cargo link replays.
add_library(swenum_res OBJECT swenum.rc)
target_include_directories(swenum_res PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_cargo_module(swenum swenum-cargo swenum.sys
    KIND kernelmodedriver
    BODY_CRATE swenum-rust
    DESTINATION reactos/system32/drivers
    EXTRA_DEPENDS swenum_res psdk)
