remove_definitions(-D_WIN32_WINNT=0x502)

# ============================================================================
# cdrom.sys — Rust cohabitation flip (11 of 13 .c -> cdrom-rust)
# ============================================================================
# The CD-ROM KMDF class driver is translated to Rust (sdk/lib/rust/cdrom-rust,
# kept on disk per the never-delete rule). 11 .c bodies (aacs/autorun/cdrom/common/
# init/ioctl/mmc/pnppower/scratch/sense/zpodd) are replaced by libcdrom_rust.a;
# data.c (the CDROM data tables the Rust modules reference) + guid.c (GUID data
# instantiation) stay C. cdrom IS boot-loaded on the LiveCD (txtsetup cdrom.sys=1,
# CriticalDeviceDatabase GenCdRom), so a finished cdrom is smoke-exercised on
# i386/amd64; arm64 boots a ramdisk so its bar is a clean link.
#
# cdrom is the fleet's only KMDF driver: every Wdf* call in the Rust dispatches
# through the kmdf-sys substrate (WdfFunctions[] table), and the C wdfdriverentry
# stub pulled in by `kmdfdriver` (FxDriverEntry) binds the framework via
# WdfVersionBind before invoking this crate's #[no_mangle] DriverEntry.
#

add_registry_inf(cdrom_reg.inf)
add_driver_inf(cdrom cdrom.inf)

# cdrom.sys is all Rust (cdrom-rust) and cargo links it:
# sdk/lib/rust/cdrom-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The translated C, kept on disk:
# 11 .c bodies replaced by Rust (cdrom-rust), kept on disk per the never-delete rule:
#   aacs.c autorun.c cdrom.c common.c init.c ioctl.c mmc.c pnppower.c scratch.c sense.c zpodd.c
# data.c's tables are Rust too (cdrom-rust/src/data, generated by `rustconv data-port`).
# guid.c stays C: its GUIDs are DEFINE_GUIDs in the headers it includes under INITGUID,
# which the data port does not emit (it writes only the TU's own declarations).
# data.c  # replaced by Rust (sdk/lib/rust/cdrom-rust/src/data/mod.rs)
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
add_library(cdrom_shims OBJECT guid.c rust_dbgprint_shim.c)
target_link_libraries(cdrom_shims PRIVATE ntoskrnl_vista libcntpr libwdfldr libntoskrnl libhal wdfdriverentry)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(cdrom_shims PRIVATE -Wno-format -Wno-unused-variable -Wno-pointer-sign)
endif()
if(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(cdrom_shims PRIVATE -Wno-unknown-pragmas -Wno-incompatible-pointer-types -Wno-switch)
endif()
if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(cdrom_shims PRIVATE -Wno-enum-conversion -Wno-tautological-constant-compare)
endif()
target_compile_definitions(cdrom_shims PRIVATE DEBUG_USE_KDPRINT
    WDF_DEVICE_NO_WDMSEC_H _WIN32_WINNT=0x603 NTDDI_VERSION=0x06030000)
add_cargo_module(cdrom cdrom-cargo cdrom.sys
    KIND kmdfdriver
    BODY_CRATE cdrom-rust
    DESTINATION reactos/system32/drivers
    NO_CAB
    EXTRA_DEPENDS cdrom_shims psdk asm)
