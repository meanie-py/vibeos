remove_definitions(-D_WIN32_WINNT=0x502)

# ============================================================================
# disk.sys — Rust cohabitation flip (all 5 C sources -> sdk/lib/rust/disk-rust)
# ============================================================================
# Every disk C source (data/disk/diskwmi/geometry/pnp) is translated to Rust
# (disk-rust, kept on disk per the never-delete rule). The PE DriverEntry comes
# from libdisk_rust.a; cmake keeps the MODULE link (set_module_type
# kernelmodedriver + add_importlibs(classpnp ntoskrnl hal)). disk.sys exports
# nothing but DriverEntry (no .spec), so this is simpler than classpnp's
# kerneldll cohab. Built on classpnp's Class* exports. Mirrors classpnp-rust.

#   data.c disk.c diskwmi.c geometry.c pnp.c  # all replaced by Rust (disk-rust); kept on disk

 # NTDDI_WIN8

add_registry_inf(disk_reg.inf)
add_driver_inf(disk disk.inf)

# disk.sys is all Rust (disk-rust) and cargo links it:
# sdk/lib/rust/disk-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# The resources are the one thing cmake still compiles, in this OBJECT library
# whose windres command the cargo link replays.
add_library(disk_res OBJECT disk.rc)
target_include_directories(disk_res PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# How cmake compiled the target's sources, now on the OBJECT libraries.
target_compile_definitions(disk_res PUBLIC
    DEBUG_USE_KDPRINT
    _WIN32_WINNT=0x602
    NTDDI_VERSION=0x06020000)
target_compile_options(disk_res PRIVATE -Wno-format -Wno-pointer-sign)
target_compile_options(disk_res PRIVATE -Wno-pointer-to-int-cast -Wno-switch)
target_compile_options(disk_res PRIVATE -Wno-unused-but-set-variable)
add_library(disk_shims OBJECT rust_cohab_stub.c)
target_link_libraries(disk_shims PRIVATE ntoskrnl_vista libcntpr wdmguid libclasspnp libntoskrnl libhal)
target_compile_definitions(disk_shims PUBLIC
    DEBUG_USE_KDPRINT
    _WIN32_WINNT=0x602
    NTDDI_VERSION=0x06020000)
target_compile_options(disk_shims PRIVATE -Wno-format -Wno-pointer-sign)
target_compile_options(disk_shims PRIVATE -Wno-pointer-to-int-cast -Wno-switch)
target_compile_options(disk_shims PRIVATE -Wno-unused-but-set-variable)
add_cargo_module(disk disk-cargo disk.sys
    KIND kernelmodedriver
    BODY_CRATE disk-rust
    DESTINATION reactos/system32/drivers
    NO_CAB
    EXTRA_DEPENDS disk_shims disk_res asm psdk)
