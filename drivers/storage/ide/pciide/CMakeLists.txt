# pciide.c is translated to Rust (pciide-rust). The ENABLE_UNIATA gate is a build
# option (upstream addition): uniata is the alternative IDE bus driver and the two
# are mutually exclusive.
#
# pciide_reg.inf is part of upstream's new "pciide is the one registered IDE driver"
# scheme, which arrives with ENABLE_UNIATA and their rewritten ATA stack. We kept
# uniata as the registered driver (see uniata/CMakeLists.txt), so registering this
# one too would put two bus drivers on the same devices.
# add_registry_inf(pciide_reg.inf)

if(NOT ENABLE_UNIATA)
    # pciide.sys is all Rust (pciide-rust) and cargo links it:
    # sdk/lib/rust/pciide-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
    # The C stays on disk, uncompiled.
    add_cargo_module(pciide pciide-cargo pciide.sys
        KIND kernelmodedriver
        BODY_CRATE pciide-rust
        DESTINATION reactos/system32/drivers
        EXTRA_DEPENDS psdk)
endif()
