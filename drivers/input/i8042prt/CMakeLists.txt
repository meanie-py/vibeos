# ============================================================================
# i8042prt.sys — cargo-driven final link (sdk/lib/rust/i8042prt-cargo via reactos-link)
# ============================================================================
# All 10 functional C sources are translated to Rust (sdk/lib/rust/i8042prt-rust,
# kept on disk per the never-delete rule); the Rust #[no_mangle] DriverEntry is
# the PE entry point. The final i8042prt.sys link is no longer cmake's
# add_library(MODULE) + rust_cohab_stub.c + the per-arch custom_command — it is
# driven by cargo (i8042prt-cargo's build.rs → reactos-link,
# ModuleKind::KernelDll, entry=DriverEntry, no exports). add_cargo_module builds
# i8042prt-rust and links it with the ntoskrnl/hal import libs (strcmp is an
# ntoskrnl export) into the shipping PE, applies the kernel-dll post-link and
# registers it on the boot media. ZERO cmake .a: libdmilib.a is gone —
# ParseSMBiosTables is now a Rust port in i8042prt-rust (crate::dmi); the host
# tool udmihelp keeps the C copy.
#
# guid.c is NOT linked: i8042prt-rust defines its sole poclass GUID
# (GUID_DEVICE_SYS_BUTTON) as a Rust-internal static used via addr_of!, so the
# C guid.c symbol is unreferenced dead residue (kept on disk).

# Registry setup is independent of the link and stays in cmake.
add_registry_inf(i8042prt_reg.inf)

# i8042prt.sys is all Rust (i8042prt-rust) and cargo links it:
# sdk/lib/rust/i8042prt-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The resources are the one thing cmake still compiles, in this OBJECT library
# whose windres command the cargo link replays.
add_library(i8042prt_res OBJECT i8042prt.rc)
target_include_directories(i8042prt_res PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_cargo_module(i8042prt i8042prt-cargo i8042prt.sys
    KIND kernelmodedriver
    BODY_CRATE i8042prt-rust
    DESTINATION reactos/system32/drivers
    EXTRA_DEPENDS i8042prt_res psdk)
