# ============================================================================
# kbdclass.sys — cargo-driven final link (sdk/lib/rust/kbdclass-cargo via reactos-link)
# ============================================================================
# The functional C sources (kbdclass.c + misc.c) were translated to Rust
# (sdk/lib/rust/kbdclass-rust, kept on disk per the never-delete rule); the
# Rust #[no_mangle] DriverEntry is the PE entry point. This cut both activates
# the Rust keyboard class driver and moves the final link off cmake onto cargo
# (kbdclass-cargo's build.rs → reactos-link, ModuleKind::KernelDll,
# entry=DriverEntry@8, no exports), linking kbdclass-rust with the ntoskrnl/hal
# import libs into the shipping PE.
#
# guid.c is NOT linked: kbdclass-rust defines its sole interface GUID
# (GUID_DEVINTERFACE_KEYBOARD) as a Rust-internal static used via addr_of!, so
# the C guid.c symbol is unreferenced dead residue (kept on disk). PSEH_LIB is
# dropped: the translation's _SEH2 handlers are dead code (DUMMY_PSEH).

# Registry setup is independent of the link and stays in cmake.
add_registry_inf(kbdclass_reg.inf)

# kbdclass.sys is all Rust (kbdclass-rust) and cargo links it:
# sdk/lib/rust/kbdclass-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The resources are the one thing cmake still compiles, in this OBJECT library
# whose windres command the cargo link replays.
add_library(kbdclass_res OBJECT kbdclass.rc)
target_include_directories(kbdclass_res PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_cargo_module(kbdclass kbdclass-cargo kbdclass.sys
    KIND kernelmodedriver
    BODY_CRATE kbdclass-rust
    DESTINATION reactos/system32/drivers
    EXTRA_DEPENDS kbdclass_res psdk)
