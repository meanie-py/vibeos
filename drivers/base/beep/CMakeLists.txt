# ============================================================================
# beep.sys — cargo-driven final link (sdk/lib/rust/beep-cargo via reactos-link)
# ============================================================================
# beep.c is translated to Rust (sdk/lib/rust/beep-rust, kept on disk per the
# never-delete rule); the Rust #[no_mangle] DriverEntry is the PE entry point.
# The final beep.sys link is no longer cmake's add_library(MODULE) — it is
# driven by cargo (beep-cargo's build.rs → reactos-link, ModuleKind::KernelDll,
# entry=DriverEntry@8, no exports), linking beep-rust with the ntoskrnl/hal
# import libs (HalMakeBeep is a hal export). The registry inf stays.

add_registry_inf(beep_reg.inf)

# beep.sys is all Rust (beep-rust) and cargo links it:
# sdk/lib/rust/beep-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The resources are the one thing cmake still compiles, in this OBJECT library
# whose windres command the cargo link replays.
add_library(beep_res OBJECT beep.rc)
target_include_directories(beep_res PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_cargo_module(beep beep-cargo beep.sys
    KIND kernelmodedriver
    BODY_CRATE beep-rust
    DESTINATION reactos/system32/drivers
    EXTRA_DEPENDS beep_res psdk)
