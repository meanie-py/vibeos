# ============================================================================
# condrv.sys — cargo-driven final link (sdk/lib/rust/condrv-cargo via reactos-link)
# ============================================================================
# condrv.c/control.c/dispatch.c are translated to Rust (sdk/lib/rust/condrv-rust,
# kept on disk per the never-delete rule); the Rust #[no_mangle] DriverEntry is
# the PE entry point. The final condrv.sys link is no longer cmake's
# add_library(MODULE) — it is driven by cargo (condrv-cargo's build.rs →
# reactos-link, ModuleKind::KernelDll, entry=DriverEntry@8, no exports).

# condrv.sys is all Rust (condrv-rust) and cargo links it:
# sdk/lib/rust/condrv-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The resources are the one thing cmake still compiles, in this OBJECT library
# whose windres command the cargo link replays.
add_library(condrv_res OBJECT condrv.rc)
target_include_directories(condrv_res PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_cargo_module(condrv condrv-cargo condrv.sys
    KIND kernelmodedriver
    BODY_CRATE condrv-rust
    DESTINATION reactos/system32/drivers
    EXTRA_DEPENDS condrv_res psdk)
