PROJECT(drwtsn32)

list(APPEND CPP_SOURCE
    # drwtsn32.cpp  # replaced by Rust (sdk/lib/rust/drwtsn32-rust/src/drwtsn32/drwtsn32_cpp_h.rs)
    # main.cpp  # replaced by Rust (sdk/lib/rust/drwtsn32-rust/src/drwtsn32/main_cpp_h.rs)
    # stacktrace.cpp  # replaced by Rust (sdk/lib/rust/drwtsn32-rust/src/drwtsn32/stacktrace_cpp_h.rs)
    # sysinfo.cpp  # replaced by Rust (sdk/lib/rust/drwtsn32-rust/src/drwtsn32/sysinfo_cpp_h.rs)
    rust_cohab_stub.cpp
    rust_varargs_shim.c
    drwtsn32.h
    precomp.h)

# drwtsn32.exe is all Rust (drwtsn32-rust) and cargo links it:
# sdk/lib/rust/drwtsn32-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
add_library(drwtsn32_shims OBJECT rust_cohab_stub.cpp rust_varargs_shim.c)
target_link_libraries(drwtsn32_shims PRIVATE udmihelp atl_classes libdbghelp libpsapi libadvapi32 libshell32 libshlwapi libmsvcrt libuser32 libkernel32 libntdll)
target_compile_definitions(drwtsn32_shims PRIVATE UNICODE _UNICODE)
set_target_cpp_properties(drwtsn32_shims WITH_EXCEPTIONS)
add_cargo_module(drwtsn32 drwtsn32-cargo drwtsn32.exe
    KIND win32exe
    BODY_CRATE drwtsn32-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS drwtsn32_shims psdk)
