# mmc_new.exe is all Rust (mmc_new-rust) and cargo links it:
# sdk/lib/rust/mmc_new-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# Generated inputs those compiles read, ordered before them.
add_library(mmc_new_shims OBJECT rust_cohab_stub.cpp)
target_link_libraries(mmc_new_shims PRIVATE uuid cpprt mmc_uuids liboleaut32 libole32 libshlwapi libshell32 libadvapi32 libcomdlg32 libcomctl32 libuser32 libgdi32 libmsvcrt libkernel32 libntdll)
target_compile_definitions(mmc_new_shims PRIVATE UNICODE _UNICODE)
add_dependencies(mmc_new_shims psdk)
set(MMC_NEW_REUSED
    ${CMAKE_BINARY_DIR}/sdk/include/psdk/libmmc_uuids.a)
add_cargo_module(mmc_new mmc_new-cargo mmc_new.exe
    KIND win32exe
    BODY_CRATE mmc_new-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS mmc_new_shims psdk ${MMC_NEW_REUSED} uuid_sources xdk)
