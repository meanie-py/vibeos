
# Console Configuration library
add_subdirectory(concfg)

include_directories(
    ${REACTOS_SOURCE_DIR}/sdk/include/reactos/subsys
    ${REACTOS_SOURCE_DIR}/win32ss/include)

spec2def(winsrv.dll winsrv.spec)

# The components (consrv + usersrv stay C — built as libconsrv.a/libusersrv.a
# component static libs the cargo link reuses).
include(consrv.cmake)
include(usersrv.cmake)

# CARGO-DRIVEN LINK + PARTIAL Rust flip (winsrv-rust). winsrv.dll's link is now
# driven by cargo (winsrv-cargo → reactos-link ModuleKind::Win32Dll, subsystem
# console:5.01) instead of cmake add_library(MODULE) — the seventh (and last of
# the SMSS/CSRSS spine) of OUR translated spine binaries moved cmake→cargo
# (after lsass + userinit + winlogon + services + smss + csrsrv). init.c is
# flipped to Rust (winsrv-rust, winsrv_dll_only: ShutdownSettings + GetTimeouts
# + DllMain); libwinsrv_rust.a is now built by cargo (.rust_staticlib) — the
# cmake add_custom_command that used to build it is retired.
#
# Axis B: the consrv/usersrv/concfg C residue (one shim/stub TU each) is now
# compiled IN-cargo via c_from_compile_db (replaying the consrv/usersrv/concfg
# cmake compile commands), and the 8 import libs are self-hosted from each .spec
# via .import_lib — so neither the component lib<x>.a nor the import lib<dll>.a
# are reused here anymore. The consrv/usersrv/concfg cmake targets stay (they
# still build their .a for compile_commands.json + other consumers; never delete
# the C). Only the genuine shared leaf/runtime statics remain reused below.

# The shared leaf/runtime .a the cargo link still reuses, as FILE paths for
# parallel-build ordering (the cargo custom target is otherwise opaque to
# cmake). winsrv.def (exports) is generated by the cargo link's spec2def.
set(WINSRV_REUSED
    ${CMAKE_CURRENT_BINARY_DIR}/concfg/libconcfg_rust.a
    ${CMAKE_BINARY_DIR}/sdk/lib/uuid/libuuid.a
    ${CMAKE_BINARY_DIR}/win32ss/libwin32ksys.a
    ${CMAKE_BINARY_DIR}/sdk/lib/crt/liblibcntpr.a
    ${CMAKE_BINARY_DIR}/sdk/lib/crt/libcntpr_rust.a
    ${CMAKE_BINARY_DIR}/subsystems/win/basesrv/CMakeFiles/libbasesrv.dir/libbasesrv.a
    ${CMAKE_BINARY_DIR}/subsystems/csr/csrsrv/CMakeFiles/libcsrsrv.dir/libcsrsrv.a)

# consrv/usersrv/concfg targets are EXTRA_DEPENDS so their compile_commands.json
# entries (which c_from_compile_db replays) are produced before the cargo link.
add_cargo_module(winsrv winsrv-cargo winsrv.dll
    KIND win32dll
    BODY_CRATE winsrv-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS consrv usersrv concfg ${WINSRV_REUSED})
