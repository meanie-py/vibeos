remove_definitions(-D_WIN32_WINNT=0x502)
add_definitions(-D_WIN32_WINNT=0x600)
include_directories(BEFORE ${REACTOS_SOURCE_DIR}/sdk/include/wine)
# Wine guards every FreeType backend body in freetype.c with SONAME_LIBFREETYPE (configure
# defines it when it finds libfreetype). FreeType is linked here, not dlopen'ed, but without
# the define the file compiles to its STATUS_NOT_IMPLEMENTED fallbacks: every glyph advance,
# bbox and outline comes back zero (Wine 11 added the guard; a sync that missed this halved
# dwrite.dll and zeroed every layout width).
add_definitions(-DSONAME_LIBFREETYPE="libfreetype")

spec2def(dwrite.dll dwrite.spec ADD_IMPORTLIB)

# dwrite.dll is C, and cargo compiles and links it:
# sdk/lib/rust/dwrite-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(dwrite_shims OBJECT shapers/arabic_table.c shapers/arabic.c analyzer.c bidi.c bracket.c direction.c font.c format.c freetype.c gdiinterop.c layout.c linebreak.c main.c mirror.c opentype.c reactos.c scripts.c shape.c)
target_link_libraries(dwrite_shims PRIVATE wine uuid oldnames freetype libshell32 libshlwapi liboleaut32 libole32 libuser32 libuser32_vista libgdi32 libmsvcrt libadvapi32 libadvapi32_vista libkernel32 libkernel32_vista libntdll)
set_wine_module_fixme(dwrite_shims) # CORE-5743: No ARRAY_SIZE macro
add_dependencies(dwrite_shims psdk)
add_cargo_module(dwrite dwrite-cargo dwrite.dll
    KIND win32dll
    DESTINATION reactos/system32
    EXTRA_DEPENDS dwrite_shims psdk uuid_sources)
