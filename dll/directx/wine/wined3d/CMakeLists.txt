# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(d3dwine_shims OBJECT rust_cohab_stub.c rust_varargs_shim.c)
target_link_libraries(d3dwine_shims PRIVATE wine vkd3d dxguid oldnames libopengl32 libuser32 libgdi32_vista libgdi32 libadvapi32 libkernel32_vista libmsvcrt libkernel32 libntdll)
set_wine_module(d3dwine_shims)
add_dependencies(d3dwine_shims wineheaders d3d_idl_headers)
add_cargo_module(d3dwine d3dwine-cargo d3dwine.dll
    KIND win32dll
    BODY_CRATE d3dwine-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS d3dwine_shims psdk d3d_idl_headers)
