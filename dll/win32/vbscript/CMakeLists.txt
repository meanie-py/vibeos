 # CORE-5743: No ARRAY_SIZE and CONST_VTABLE

# vbscript.dll is C, and cargo compiles and links it:
# sdk/lib/rust/vbscript-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(vbscript_shims OBJECT compile.c global.c interp.c lex.c regexp.c utils.c vbdisp.c vbregexp.c vbscript.c vbscript_main.c ${CMAKE_CURRENT_BINARY_DIR}/parser.tab.c)
target_link_libraries(vbscript_shims PRIVATE uuid wine oldnames liboleaut32 libole32 libuser32 libmsvcrt libkernel32 libntdll)
set_wine_module_fixme(vbscript_shims)
add_dependencies(vbscript_shims vbscript_idlheader stdole2)
add_cargo_module(vbscript vbscript-cargo vbscript.dll
    KIND win32dll
    DESTINATION reactos/system32
    EXTRA_DEPENDS vbscript_shims psdk ${CMAKE_CURRENT_BINARY_DIR}/vbsglobal.tlb ${CMAKE_CURRENT_BINARY_DIR}/vbsregexp10.tlb ${CMAKE_CURRENT_BINARY_DIR}/vbsregexp55.tlb uuid_sources)
