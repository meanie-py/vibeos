    # client.c  # replaced by Rust (sdk/lib/rust/oleacc-rust/src/oleacc/client.rs)
    # main.c  # replaced by Rust (sdk/lib/rust/oleacc-rust/src/oleacc/main.rs)
    # propservice.c  # replaced by Rust (sdk/lib/rust/oleacc-rust/src/oleacc/propservice.rs)
    # window.c  # replaced by Rust (sdk/lib/rust/oleacc-rust/src/oleacc/window.rs)
    )

add_idl_headers(oleacc_idlheader oleacc_classes.idl)

add_idl_reg_objects(oleacc registry oleacc_classes.idl)
add_idl_reg_objects(oleacc regtypelib oleacc_classes.idl)
 # CORE-5743: No ARRAY_SIZE and CONST_VTABLE

# oleacc.dll is all Rust (oleacc-rust) and cargo links it:
# sdk/lib/rust/oleacc-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The translated C, kept on disk:
# guid.c  # replaced by Rust (sdk/lib/rust/oleacc-rust/src/oleacc/guid.rs)
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(oleacc_shims OBJECT rust_cohab_stub.c ${CMAKE_CURRENT_BINARY_DIR}/oleacc_classes_p.c ${CMAKE_CURRENT_BINARY_DIR}/proxy.dlldata.c)
target_link_libraries(oleacc_shims PRIVATE uuid wine oldnames liboleaut32 libole32 libuser32 librpcrt4 libmsvcrt libkernel32 libntdll)
set_wine_module_fixme(oleacc_shims)
add_dependencies(oleacc_shims oleacc_idlheader stdole2)
set(OLEACC_REUSED
    ${CMAKE_BINARY_DIR}/dll/win32/oleacc/oleacc_classes_r.o
    ${CMAKE_BINARY_DIR}/dll/win32/oleacc/oleacc_classes_t.o)
add_cargo_module(oleacc oleacc-cargo oleacc.dll
    KIND win32dll
    BODY_CRATE oleacc-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS oleacc_shims psdk oleacc_r oleacc_t ${OLEACC_REUSED} uuid_sources)
