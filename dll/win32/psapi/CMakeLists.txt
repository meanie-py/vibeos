spec2def(psapi.dll psapi.spec ADD_IMPORTLIB)

# psapi.c is Rust now (psapi-rust, `rustconv port`); the C stays on disk. psapi_static is an
# INTERFACE library over the Rust archive (`cargo-cell --interface`) -- kernel32 and
# kernelbase link it too.
# psapi.c  # replaced by Rust (sdk/lib/rust/psapi-rust/src/psapi/psapi.rs)
rust_staticlib_variant(psapi_rust psapi-rust "" TRUE)
add_library(psapi_static INTERFACE)
add_dependencies(psapi_static psdk rust_psapi_rust)
target_link_libraries(psapi_static INTERFACE $<LINK_ONLY:${psapi_rust_LIB}>)

# psapi.dll is all Rust (psapi-rust) and cargo links it:
# sdk/lib/rust/psapi-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The translated C, kept on disk:
# The DLL target is all Rust too: main.c's DllMain is psapi-rust's `dllmain` feature
# (src/psapi/main.rs), built only for this DLL so kernel32/kernelbase, which link
# psapi_static, get no second entry point. The C stays on disk.
# main.c  # replaced by Rust (sdk/lib/rust/psapi-rust/src/psapi/main.rs)
add_cargo_module(psapi psapi-cargo psapi.dll
    KIND win32dll
    BODY_CRATE psapi-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS psdk)
