# <libxslt/...> used to resolve through the global sdk/include/reactos/libs root.
include_directories(${REACTOS_SOURCE_DIR}/vendor/libxslt/include)

add_definitions(
    -D__ROS_LONG64__
    -D_WINE
    -DLIBXML_STATIC
    -DCOM_NO_WINDOWS_H)

spec2def(msxml3.dll msxml3.spec ADD_IMPORTLIB NO_PRIVATE_WARNINGS)
add_typelib(msxml3_v1.idl)

FLEX_TARGET(xslpattern_scanner xslpattern.l ${CMAKE_CURRENT_BINARY_DIR}/xslpattern.yy.c)
BISON_TARGET(xslpattern_parser xslpattern.y ${CMAKE_CURRENT_BINARY_DIR}/xslpattern.tab.c COMPILE_FLAGS "-p xslpattern_")
ADD_FLEX_BISON_DEPENDENCY(xslpattern_scanner xslpattern_parser)

list(APPEND msxml3_rc_deps
    ${CMAKE_CURRENT_SOURCE_DIR}/msxml3.manifest
    ${CMAKE_CURRENT_SOURCE_DIR}/msxml3_v1.rgs
    ${CMAKE_CURRENT_SOURCE_DIR}/xmlparser.rgs
    ${CMAKE_CURRENT_BINARY_DIR}/msxml3_v1.tlb)

add_idl_headers(xmlparser_idlheader xmlparser.idl)

 # CORE-5743: No ARRAY_SIZE and CONST_VTABLE

# msxml3.dll is C, and cargo compiles and links it:
# sdk/lib/rust/msxml3-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(msxml3_shims OBJECT attribute.c bsc.c cdata.c comment.c dispex.c docfrag.c doctype.c domdoc.c domimpl.c element.c entityref.c httprequest.c main.c mxnamespace.c mxwriter.c node.c nodelist.c nodemap.c parseerror.c pi.c reactos.c saxreader.c schema.c selection.c stylesheet.c text.c xdr.c xmldoc.c xmlelem.c xmlparser.c xmlview.c factory.c uuid.c ${CMAKE_CURRENT_BINARY_DIR}/xslpattern.yy.c ${CMAKE_CURRENT_BINARY_DIR}/xslpattern.tab.c)
target_link_libraries(msxml3_shims PRIVATE libxml2 iconv-static uuid wine wine_dll_register zlib liblibxslt liburlmon libws2_32 libshlwapi libbcrypt liboleaut32 libole32 libuser32 libmsvcrt libkernel32 libntdll)
if(CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(msxml3_shims PRIVATE -Wno-incompatible-function-pointer-types)
endif()
set_wine_module_fixme(msxml3_shims)
add_dependencies(msxml3_shims xmlparser_idlheader stdole2)
set(MSXML3_REUSED
    ${CMAKE_BINARY_DIR}/sdk/lib/3rdparty/libwin-iconv/libiconv.a)
add_cargo_module(msxml3 msxml3-cargo msxml3.dll
    KIND win32dll
    DESTINATION reactos/system32
    EXTRA_DEPENDS msxml3_shims psdk ${CMAKE_CURRENT_BINARY_DIR}/msxml3_v1.tlb ${MSXML3_REUSED} uuid_sources)
