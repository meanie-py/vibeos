# --- Rust cohabitation (setupapi-rust) ------------------------------------
# libsetupapi_rust.a carries every TU of the module: cfgmgr.c (whose RpcTryExcept guards
# compile to nothing under DUMMY_PSEH and are real rpc_try calls in Rust), then the other
# seventeen. Only the generated pnp_c.c / setupapi_stubs.c are still C.

# --------------------------------------------------------------------------

include(c_oracle.cmake)  # the pre-flip C as <target>_c: python -m rustconv.c_oracle

# setupapi.dll is all Rust (setupapi-rust) and cargo links it:
# sdk/lib/rust/setupapi-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
add_library(setupapi_shims OBJECT ${CMAKE_CURRENT_BINARY_DIR}/pnp_c.c)
target_link_libraries(setupapi_shims PRIVATE uuid wine oldnames libgdi32 libcomctl32 libadvapi32 libuser32 librpcrt4 libversion libmsvcrt libkernel32 libntdll)
target_compile_definitions(setupapi_shims PRIVATE UNICODE _UNICODE)
add_cargo_module(setupapi setupapi-cargo setupapi.dll
    KIND win32dll
    BODY_CRATE setupapi-rust
    DESTINATION reactos/system32
    DELAY_IMPORTS
    EXTRA_DEPENDS setupapi_shims psdk uuid_sources)
