add_definitions(
    -D_URLMON_
    -DENTRY_PREFIX=URLMON_
    -DPROXY_DELEGATION
    -DWINE_REGISTER_DLL)

# PROXY_CLSID_IS contains braces and commas which cannot be passed via -D
# on all compilers/shells. Write it to a header and force-include it.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/proxy_clsid.h"
    "#define PROXY_CLSID_IS {0x79EAC9F1,0xBAF9,0x11CE,{0x8C,0x82,0x00,0xAA,0x00,0x4B,0xA9,0x0B}}\n")
add_compile_options(-include "${CMAKE_CURRENT_BINARY_DIR}/proxy_clsid.h")

spec2def(urlmon.dll urlmon.spec ADD_IMPORTLIB)
add_rpcproxy_files(urlmon_urlmon.idl)

set(_source_file ${CMAKE_CURRENT_SOURCE_DIR}/urlmon.inf)
set(_converted_file ${CMAKE_CURRENT_BINARY_DIR}/urlmon_utf16.inf)
utf16le_convert(${_source_file} ${_converted_file})

add_idl_reg_objects(urlmon registry urlmon_urlmon.idl)

# Upstream solves the "CMake cannot escape a GUID on the command line" problem with a
# second `-include clsid.h`. We already solve it by GENERATING proxy_clsid.h above and
# -include'ing that, and a second -include is actively harmful here: CMake de-duplicates
# repeated COMPILE_OPTIONS, so the second `-include` flag is dropped and its path becomes
# a second INPUT file — "clang: error: cannot specify -o when generating multiple output
# files", on every TU in this module. Upstream's branch is therefore dropped, not merged.

# urlmon.dll is C, and cargo compiles and links it:
# sdk/lib/rust/urlmon-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
add_library(urlmon_shims OBJECT axinstall.c bindctx.c binding.c bindprot.c download.c file.c format.c ftp.c gopher.c http.c internet.c mimefilter.c mk.c protocol.c sec_mgr.c session.c umon.c umstream.c uri.c urlmon_main.c usrmarshal.c ${CMAKE_CURRENT_BINARY_DIR}/proxy.dlldata.c ${CMAKE_CURRENT_BINARY_DIR}/urlmon_urlmon_p.c)
target_link_libraries(urlmon_shims PRIVATE msvcrtex uuid wine oldnames librpcrt4 libpropsys libole32 liboleaut32 libshlwapi libshell32 libwininet libnormaliz libuser32 libadvapi32 libkernel32_vista libmsvcrt libkernel32 libntdll)
set_wine_module_fixme(urlmon_shims)
set(URLMON_REUSED
    ${CMAKE_BINARY_DIR}/dll/win32/urlmon/urlmon_urlmon_r.o)
add_cargo_module(urlmon urlmon-cargo urlmon.dll
    KIND win32dll
    DESTINATION reactos/system32
    DELAY_IMPORTS
    EXTRA_DEPENDS urlmon_shims psdk ${CMAKE_CURRENT_BINARY_DIR}/urlmon_utf16.inf urlmon_r ${URLMON_REUSED} uuid_sources)
