# --- Rust cohabitation (advapi32-rust) ------------------------------------
# Build libadvapi32_rust.a from the workspace crate (STANDARD user-mode per-arch
# targets; i386 gets -sse,-sse2 via sdk/lib/rust/.cargo/config.toml). The
# #[no_mangle] bodies replace the commented-out .c TUs above; lld first-match
# (--allow-multiple-definition) lets the Rust staticlib's panic_handler/
# eh_personality coexist with whatever the C cohab pulls in.

# --------------------------------------------------------------------------

# advapi32.dll is all Rust (advapi32-rust) and cargo links it:
# sdk/lib/rust/advapi32-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
add_library(advapi32_shims OBJECT ${CMAKE_CURRENT_BINARY_DIR}/eventlogrpc_c.c ${CMAKE_CURRENT_BINARY_DIR}/lsa_c.c ${CMAKE_CURRENT_BINARY_DIR}/seclogon_c.c ${CMAKE_CURRENT_BINARY_DIR}/svcctl_c.c ${CMAKE_CURRENT_BINARY_DIR}/winreg_c.c)
target_link_libraries(advapi32_shims PRIVATE cryptlib wine etwtrace libadvapi32_vista librpcrt4 libkernel32 libntdll)
target_compile_definitions(advapi32_shims PRIVATE UNICODE _UNICODE)
add_cargo_module(advapi32 advapi32-cargo advapi32.dll
    KIND win32dll
    BODY_CRATE advapi32-rust
    DESTINATION reactos/system32
    DELAY_IMPORTS
    EXTRA_DEPENDS advapi32_shims psdk)
