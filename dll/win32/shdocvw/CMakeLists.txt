add_definitions(
    -D_SHDOCVW_)

spec2def(shdocvw.dll shdocvw.spec ADD_IMPORTLIB)

add_typelib(res/shdocvw_v1.idl)

add_library(shdocvw_sublib OBJECT
    CExplorerBand.cpp
    CFavBand.cpp
    CNSCBand.cpp
    mrulist.cpp
    objects.cpp
    utility.cpp
    winlist.cpp)
target_link_libraries(shdocvw_sublib PRIVATE atl_classes)
target_compile_definitions(shdocvw_sublib PRIVATE $<TARGET_PROPERTY:shdocvw_shims,COMPILE_DEFINITIONS>)
target_compile_options(shdocvw_sublib PRIVATE $<TARGET_PROPERTY:shdocvw_shims,COMPILE_OPTIONS>)
add_dependencies(shdocvw_sublib psdk)

# shdocvw.dll is C, and cargo compiles and links it:
# sdk/lib/rust/shdocvw-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(shdocvw_shims OBJECT shdocvw_main.c shlinstobj.c)
target_link_libraries(shdocvw_shims PRIVATE cpprt uuid wine libshlwapi libadvapi32 libmsvcrt libcomctl32 libkernel32 libntdll)
target_compile_definitions(shdocvw_shims PRIVATE UNICODE _UNICODE)
target_include_directories(shdocvw_shims BEFORE PRIVATE
    ${REACTOS_SOURCE_DIR}/sdk/include/wine)
add_dependencies(shdocvw_shims stdole2)
add_cargo_module(shdocvw shdocvw-cargo shdocvw.dll
    KIND win32dll
    DESTINATION reactos/system32
    DELAY_IMPORTS
    EXTRA_DEPENDS shdocvw_shims psdk ${CMAKE_CURRENT_BINARY_DIR}/shdocvw_v1.tlb uuid_sources xdk shdocvw_sublib)
