# Native SEH for the TUs whose __TRY/__EXCEPT is the RPC error contract
# (dummy-pseh-verdict.md, "Consequence for RPC clients"). Under the global DUMMY_PSEH
# mode none of these handlers runs: a failed COM/RPC call's RpcRaiseException is
# resumed by the dispatcher, RaiseException returns, and RpcRaiseException ends in
# ExitProcess(1), so any client of a dead or refusing server died (Wine's ole32
# marshal test: a proxy called from the wrong apartment). Every try body here is
# call-shaped (the raise lands in a callee), the same contract as the advapi32
# EventLog shim. They skip the PCH, which would otherwise pin pseh2.h to DUMMY.
# i386 keeps DUMMY: clang i386 native SEH is unreliable.
set(RPCRT4_SEH_SOURCE
    cstub.c
    ndr_stubless.c
    rpc_server.c)
if(ARCH STREQUAL "amd64" OR ARCH STREQUAL "arm64")
    set_source_files_properties(${RPCRT4_SEH_SOURCE} PROPERTIES COMPILE_DEFINITIONS _USE_NATIVE_SEH)
endif()

# rpcrt4.dll is C, and cargo compiles and links it:
# sdk/lib/rust/rpcrt4-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(rpcrt4_shims OBJECT cproxy.c cpsf.c ndr_clientserver.c ndr_contexthandle.c ndr_es.c ndr_fullpointer.c ndr_marshall.c ndr_ole.c ros_extra.c rpc_assoc.c rpc_async.c rpc_binding.c rpc_callattr.c rpc_message.c rpcrt4_main.c rpc_transport.c thunks.c unix_func.c ${CMAKE_CURRENT_BINARY_DIR}/epm_c.c cstub.c ndr_stubless.c rpc_server.c rpc_epmap.c ${CMAKE_CURRENT_BINARY_DIR}/epm_guard.c ndr_typelib.c ${CMAKE_CURRENT_BINARY_DIR}/ndr_types_p.c ${CMAKE_CURRENT_BINARY_DIR}/proxy.dlldata.c)
target_link_libraries(rpcrt4_shims PRIVATE wine wine_dll_register uuid oldnames libadvapi32 libadvapi32_vista libkernel32_vista libws2_32 libmsvcrt libkernel32 libntdll)
set_wine_module_fixme(rpcrt4_shims) # CORE-5743: No ARRAY_SIZE and CONST_VTABLE
add_dependencies(rpcrt4_shims ndr_types_header)
add_cargo_module(rpcrt4 rpcrt4-cargo rpcrt4.dll
    KIND win32dll
    DESTINATION reactos/system32
    DELAY_IMPORTS
    EXTRA_DEPENDS rpcrt4_shims psdk uuid_sources)
