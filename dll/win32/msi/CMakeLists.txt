remove_definitions(-D_WIN32_WINNT=0x502)
add_definitions(-D_WIN32_WINNT=0x600)

add_definitions(-D__WINESRC__ -DMSIRUNMODE=MSIRUNMODE_T)
spec2def(msi.dll msi.spec ADD_IMPORTLIB)
generate_idl_iids(msiserver.idl)
generate_idl_iids(winemsi.idl)

BISON_TARGET(cond cond.y ${CMAKE_CURRENT_BINARY_DIR}/cond.tab.c)
BISON_TARGET(sql sql.y ${CMAKE_CURRENT_BINARY_DIR}/sql.tab.c)

set(OLD_IDL_FLAGS ${IDL_FLAGS})
set(IDL_FLAGS ${IDL_FLAGS} --prefix-server=s_)
add_rpc_files(client winemsi.idl)
# The guarded client stubs (remote_*_guarded, winemsi_guard.c) run under RtlSehTryExcept, since
# __TRY/__EXCEPT(rpc_filter) is a no-op under DUMMY_PSEH on every arch.
add_rpc_guards(winemsi windef.h winbase.h)
add_rpc_files(server winemsi.idl)
add_idl_headers(msi_idlheader msiserver.idl winemsi.idl)
set(IDL_FLAGS ${OLD_IDL_FLAGS})

if(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_ID STREQUAL "Clang")
 # error: passing argument 1 of 'RtlGetVersion' from incompatible pointer type
endif()

add_idl_reg_objects(msi regtypelib msiserver.idl)
add_idl_reg_objects(msi registry msiserver.idl)

# msi.dll is C, and cargo compiles and links it:
# sdk/lib/rust/msi-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(msi_shims OBJECT action.c alter.c appsearch.c assembly.c automation.c classes.c create.c database.c delete.c dialog.c distinct.c drop.c files.c font.c format.c handle.c insert.c install.c media.c msi.c msi_main.c msiquery.c package.c patch.c record.c registry.c script.c select.c source.c storages.c streams.c string.c table.c tokenize.c update.c upgrade.c where.c custom.c suminfo.c ${CMAKE_CURRENT_BINARY_DIR}/cond.tab.c ${CMAKE_CURRENT_BINARY_DIR}/sql.tab.c ${CMAKE_CURRENT_BINARY_DIR}/msiserver_i.c ${CMAKE_CURRENT_BINARY_DIR}/winemsi_c.c ${CMAKE_CURRENT_BINARY_DIR}/winemsi_s.c ${CMAKE_CURRENT_BINARY_DIR}/winemsi_guard.c)
target_link_libraries(msi_shims PRIVATE uuid wine_dll_register oldnames libadvapi32 libadvapi32_vista libcabinet libcomctl32 libgdi32 libole32 liboleaut32 libshell32 libshlwapi librpcrt4 liburlmon libuser32 libversion libwininet libsxs libimagehlp libmspatcha libmsvcrt libkernel32_vista libkernel32 libntdll)
if(CMAKE_C_COMPILER_ID STREQUAL "GNU" OR CMAKE_C_COMPILER_ID STREQUAL "Clang")
    target_compile_options(msi_shims PRIVATE -Wno-incompatible-pointer-types)
endif()
add_dependencies(msi_shims msi_idlheader)
set(MSI_REUSED
    ${CMAKE_BINARY_DIR}/dll/win32/msi/msiserver_r.o
    ${CMAKE_BINARY_DIR}/dll/win32/msi/msiserver_t.o)
add_cargo_module(msi msi-cargo msi.dll
    KIND win32dll
    DESTINATION reactos/system32
    DELAY_IMPORTS
    EXTRA_DEPENDS msi_shims psdk msi_t msi_r ${MSI_REUSED} uuid_sources)
