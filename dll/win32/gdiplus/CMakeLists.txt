# gdiplus.dll is C, and cargo compiles and links it:
# sdk/lib/rust/gdiplus-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
add_library(gdiplus_shims OBJECT brush.c customlinecap.c font.c gdiplus.c graphics.c graphicspath.c image.c imageattributes.c matrix.c metafile.c pathiterator.c pen.c region.c stringformat.c guid.c)
target_link_libraries(gdiplus_shims PRIVATE wine libmlang libmsvcrt libshlwapi liboleaut32 libole32 libuser32 libgdi32 libkernel32 libkernel32_vista libntdll libntdll_vista)
target_compile_options(gdiplus_shims PRIVATE -Wno-incompatible-pointer-types)
set_wine_module_fixme(gdiplus_shims)
add_cargo_module(gdiplus gdiplus-cargo gdiplus.dll
    KIND win32dll
    DESTINATION reactos/system32 reactos/winsxs/${WINARCH}_microsoft.windows.gdiplus_6595b64144ccf1df_1.1.7601.23038_none_deadbeef reactos/winsxs/${WINARCH}_microsoft.windows.gdiplus_6595b64144ccf1df_1.0.14393.0_none_deadbeef
    DELAY_IMPORTS
    EXTRA_DEPENDS gdiplus_shims psdk)
