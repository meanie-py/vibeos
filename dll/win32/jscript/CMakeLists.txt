spec2def(jscript.dll jscript.spec)

# Preserve the original C source; the Rust archive supplies decode_source.

BISON_TARGET(cc_parser cc_parser.y ${CMAKE_CURRENT_BINARY_DIR}/cc_parser.tab.c)
BISON_TARGET(parser parser.y ${CMAKE_CURRENT_BINARY_DIR}/parser.tab.c)

list(APPEND jscript_rc_deps
    ${CMAKE_CURRENT_SOURCE_DIR}/jscript.rgs)

add_idl_headers(jscript_idlheader jscript_classes.idl)
add_idl_headers(jscript_jsdisp_header jsdisp.idl)
add_idl_reg_objects(jscript registry jscript_classes.idl)
add_idl_reg_objects(jscript regtypelib jsglobal.idl)

# jscript.dll is all Rust (jscript-rust) and cargo links it:
# sdk/lib/rust/jscript-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C stays on disk, uncompiled.
# The translated C, kept on disk:
# decode.c (compiler-translated into jscript-rust/src/decode.rs)
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
# Generated inputs those compiles read, ordered before them.
add_library(jscript_shims OBJECT activex.c array.c arraybuf.c bool.c compile.c date.c dispex.c engine.c enumerator.c error.c function.c global.c jscript.c jscript_main.c json.c jsregexp.c jsstr.c jsutils.c lex.c math.c number.c object.c regexp.c set.c string.c vbarray.c guid.c ${CMAKE_CURRENT_BINARY_DIR}/cc_parser.tab.c ${CMAKE_CURRENT_BINARY_DIR}/parser.tab.c)
target_link_libraries(jscript_shims PRIVATE uuid wine oldnames wine_dll_register libuser32 libole32 liboleaut32 libadvapi32 libmsvcrt libkernel32 libntdll)
set_wine_module_fixme(jscript_shims) # CORE-5743: No ARRAY_SIZE nor CONST_VTABLE
add_dependencies(jscript_shims jscript_idlheader jscript_jsdisp_header stdole2)
set(JSCRIPT_REUSED
    ${CMAKE_BINARY_DIR}/dll/win32/jscript/jscript_classes_r.o
    ${CMAKE_BINARY_DIR}/dll/win32/jscript/jsglobal_t.o)
add_cargo_module(jscript jscript-cargo jscript.dll
    KIND win32dll
    BODY_CRATE jscript-rust
    DESTINATION reactos/system32
    EXTRA_DEPENDS jscript_shims psdk jscript_r jscript_t ${JSCRIPT_REUSED} uuid_sources)
