# upstream dropped these, relying on the vendored libs publishing their headers through
# their own targets. Our libpng/libjpeg/libtiff are at different versions (we declined the
# libjpeg v10 API break and are AHEAD on libpng) and do not, so the paths stay spelled out.
include_directories(
    ${REACTOS_SOURCE_DIR}/vendor/libjpeg/include
    ${REACTOS_SOURCE_DIR}/vendor/zlib/include
    ${REACTOS_SOURCE_DIR}/vendor/libpng/include
    ${REACTOS_SOURCE_DIR}/vendor/libtiff/include)

add_rpcproxy_files(windowscodecs_wincodec.idl)

add_idl_reg_objects(windowscodecs registry windowscodecs_wincodec.idl)
 # CORE-5743: No CONST_VTABLE

# windowscodecs.dll is C, and cargo compiles and links it:
# sdk/lib/rust/windowscodecs-cargo (reactos-link, generated by tools/cargo_cell_gen.py).
# The C still compiled, in an OBJECT library whose compile the cargo link replays.
# How cmake compiled the target's sources, now on the OBJECT libraries.
add_library(windowscodecs_shims OBJECT bmpdecode.c bmpencode.c clipper.c clsfactory.c colorcontext.c colortransform.c converter.c ddsformat.c decoder.c encoder.c fliprotate.c gifformat.c icoformat.c imgfactory.c info.c libjpeg.c libpng.c libtiff.c main.c metadatahandler.c metadataquery.c palette.c pngformat.c propertybag.c proxy.c regsvr.c scaler.c stream.c tgaformat.c ungif.c uuid.c wincodecs_common.c bitmap.c guid.c ${CMAKE_CURRENT_BINARY_DIR}/proxy.dlldata.c ${CMAKE_CURRENT_BINARY_DIR}/windowscodecs_wincodec_p.c)
target_link_libraries(windowscodecs_shims PRIVATE wine uuid oldnames liblibjpeg liblibpng liblibtiff libole32 liboleaut32 librpcrt4 libshlwapi libuser32 libgdi32 libadvapi32 libadvapi32_vista libpropsys libmsvcrt libkernel32 libntdll)
set_wine_module_fixme(windowscodecs_shims)
set(WINDOWSCODECS_REUSED
    ${CMAKE_BINARY_DIR}/dll/win32/windowscodecs/windowscodecs_wincodec_r.o)
add_cargo_module(windowscodecs windowscodecs-cargo windowscodecs.dll
    KIND win32dll
    DESTINATION reactos/system32
    EXTRA_DEPENDS windowscodecs_shims psdk windowscodecs_r ${WINDOWSCODECS_REUSED} uuid_sources)
